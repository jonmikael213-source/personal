        // displaying a destination choices with another method inside for taking an input purposes
        public static int chooseDestination() {

            // displaying choices
            String message = "CHOOSE YOUR DESTINATION\n\n";
            for (int i = 0; i < destinations.length; i++) {
                message += (i + 1) + ". " + destinations[i] + "\n";
            }

            JOptionPane.showMessageDialog(null, message);

            // after displaying, will call another method for inputing
            return inputInt(
                "Enter destination number:",
                1, destinations.length
            ) - 1;
        }

        static int inputInt(String message, int minimum, int maximum) {
            // looping so that it will run until it takes a right input
            while (true) {
                // using try and catch to not let the program crash if the user input completely wrong answer
                try {
                    int number = Integer.parseInt(input(message));

                    if (number >= minimum && number <= maximum) {
                        // automatically stop the loop
                        return number;
                    }
                    // if user input low or high number compare to the choices
                    JOptionPane.showMessageDialog(
                        null,
                        "Please enter a number from " + minimum + " to " + maximum + "."
                    );
                // if user did not input correct to what is needed
                } catch (NumberFormatException e) {
                    JOptionPane.showMessageDialog(
                        null,
                        "Invalid input. Please enter a whole number."
                    );
                }
            }
        }
        // only for taking and returing a string
        static String input(String message) {
            String answer = JOptionPane.showInputDialog(
                null,
                message,
                "LanIya Star Hotel",
                JOptionPane.QUESTION_MESSAGE
            );

            // this will stop the program if the user presses Cancel.
            if (answer == null) {
                System.exit(0);
            }
            // removing unnecessary spaces in front and back that may result an error
            return answer.trim();
        }
